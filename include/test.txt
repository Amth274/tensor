function graph_build_backward(g):

    initialize g.grad_of to INVALID

    // Seed output gradients
    for each tensor t:
        if t.is_output:
            seed = create_seed_tensor(t)
            g.grad_of[t.id] = seed

    // Reverse traversal
    for i = execution_count - 1 down to 0:

        node = g.nodes[g.execution_order[i]]

        out = node.outputs[0]
        grad_out = g.grad_of[out]

        if grad_out == INVALID:
            continue

        switch node.op:

            case OP_ADD:
                x = node.inputs[0]
                y = node.inputs[1]

                accumulate(x, grad_out)
                accumulate(y, grad_out)

            case OP_MUL:
                x = node.inputs[0]
                y = node.inputs[1]

                dx = create_node(OP_MUL, grad_out, y)
                dy = create_node(OP_MUL, grad_out, x)

                accumulate(x, dx)
                accumulate(y, dy)
